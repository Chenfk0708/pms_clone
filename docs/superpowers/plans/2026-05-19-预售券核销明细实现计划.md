export async function fetchPreSaleCouponMallDashboard(query: PreSaleCouponMallQuery, signal?: AbortSignal) {
  const normalized = normalizeQuery(query)
  const provider = resolvePreSaleCouponMallProvider()
  if (provider === 'api') {
    throw new PreSaleCouponMallServiceError(
      '预售券核销明细加载失败，请稍后重试',
      envelope(503, 'service unavailable', null, 'api-pre-sale-coupon-mall-unavailable'),
      normalized,
    )
  }
  await waitForMockLatency(signal)
  if (normalized.state === 'error') {
    throw new PreSaleCouponMallServiceError(
      '预售券核销明细加载失败，请稍后重试',
      envelope(503, 'pre sale coupon mall query failed', null, 'mock-pre-sale-coupon-mall-error-001'),
      normalized,
    )
  }
  return adaptDashboard(provider, normalized, makeDashboardEnvelope(normalized))
}
